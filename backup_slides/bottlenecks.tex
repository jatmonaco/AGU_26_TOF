\begin{frame}[label=bottlenecks]
    \frametitle{My Focus Areas}
    \uncover<1->{\alert{Better uncertainty quantification} yields better science
    \begin{itemize}
        \item[$\blacktriangleright$] Measurement precision is often unquantified
        \item[$\blacktriangleright$] Uncertainties inform how much to trust the data
    \end{itemize}}
    \uncover<2->{\alert{Better analytical techniques} to extract more information from imperfect data
    \begin{itemize}
        \item[$\blacktriangleright$] Each observation provides partial measurements 
        \item[$\blacktriangleright$] Difficult to analyze measurements at random times and locations
        \item[$\blacktriangleright$] Estimating different behaviors require different analytical techniques
    \end{itemize}}
    \uncover<3->{Creating methods and technologies for \alert{easy wide-scale network deployments}
    \begin{itemize}
        \item[$\blacktriangleright$] Understanding global dynamics require global observations
        \item[$\blacktriangleright$] Difficult to deploy new instruments
        \item[$\blacktriangleright$] No established tools to determine the placement that best support science
    \end{itemize}}
\end{frame}
